// --- MANUAL SCALPING LOGIC ---
window.executeScalping = async function(action) {
    const lotInput = document.getElementById('scalp-lot');
    if (!lotInput) return;
    
    const volume = parseFloat(lotInput.value);
    if (isNaN(volume) || volume <= 0) {
        showToast('Lot size tidak valid', 'error');
        return;
    }
    
    try {
        const res = await fetch('/api/manual/execute', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ action: action, volume: volume })
        });
        const data = await res.json();
        if (data.status === 'success') {
            showToast(`Sukses Eksekusi ${action} ${volume} Lot! (Ticket: ${data.ticket})`, 'success');
        } else {
            showToast(data.message, 'error');
        }
    } catch (e) {
        console.error(e);
        showToast('Error executing order', 'error');
    }
};

window.closeManualTrade = async function(ticket) {
    try {
        const res = await fetch('/api/manual/close', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ ticket: parseInt(ticket) })
        });
        const data = await res.json();
        if (data.status === 'success') {
            showToast(data.message, 'success');
        } else {
            showToast(data.message, 'error');
        }
    } catch (e) {
        console.error(e);
        showToast('Error closing order', 'error');
    }
};

window.closeAllManual = async function() {
    // For simplicity, we can fetch status first, then close them sequentially
    try {
        const res = await fetch('/api/manual/status');
        const data = await res.json();
        if (data.status === 'success' && data.active_positions.length > 0) {
            for (let p of data.active_positions) {
                await window.closeManualTrade(p.ticket);
            }
            showToast('Semua posisi manual ditutup!', 'success');
        } else {
            showToast('Tidak ada posisi manual untuk ditutup', 'warning');
        }
    } catch(e) {
        console.error(e);
    }
};
